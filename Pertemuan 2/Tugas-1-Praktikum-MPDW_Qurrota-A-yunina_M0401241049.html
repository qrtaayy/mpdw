# Impor data 100 jam yang sudah dirata-ratakan
data1 <- read.csv ("C://Users//USER//OneDrive//ドキュメント//MPDW 1//mpdw//Tugas 1 Praktikum MPDW//Data Suhu 100 Pertama.csv")

colnames(data1)
## [1] "Date.Time" "T..degC."
# Mengubah ke objek time series
data1.ts <- ts(data1$`T..degC`)
# Bagi data latih (80% = 80 jam) dan data uji (20% = 20 jam)
n <- nrow(data1)
n_train <- floor(0.8 * n)

training_ma <- data1[1:n_train, ]
testing_ma  <- data1[(n_train+1):n, ]

train_ma.ts <- ts(training_ma$`T..degC`)
test_ma.ts  <- ts(testing_ma$`T..degC`)

# 1. Single Moving Average (SMA n=4)
data.sma <- SMA(train_ma.ts, n = 4)
data.ramal <- c(NA, data.sma)
n_test <- length(test_ma.ts)

data.gab <- cbind(
  aktual = c(data1.ts),
  pemulusan = c(data.sma, rep(NA, n_test)),
  ramalan = c(data.ramal, rep(data.ramal[length(data.ramal)], n_test - 1))
)

# 2. Double Moving Average (DMA n=4)
dma <- SMA(data.sma, n = 4)
At <- 2*data.sma - dma
Bt <- 2/(4-1)*(data.sma - dma)
data.dma <- At + Bt
data.ramal2 <- c(NA, data.dma)

f <- c()
for (i in 1:n_test) {
  f[i] <- At[length(At)] + Bt[length(Bt)] * i
}

data.gab2 <- cbind(
  aktual = c(data1.ts),
  pemulusan1 = c(data.sma, rep(NA, n_test)),
  pemulusan2 = c(dma, rep(NA, n_test)),
  At = c(At, rep(NA, n_test)),
  Bt = c(Bt, rep(NA, n_test)),
  ramalan = c(data.ramal2, f[-1])
)

# 3. Single Exponential Smoothing (SES Optimum)
HWopt <- HoltWinters(train_ma.ts, gamma = FALSE, beta = FALSE, alpha = NULL)
ramalanopt <- forecast(HWopt, h = n_test)

# 4. Double Exponential Smoothing (DES Optimum)
des.opt <- HoltWinters(train_ma.ts, gamma = FALSE, alpha = NULL, beta = NULL)
ramalandesopt <- forecast(des.opt, h = n_test)

# Hitung Akurasi Data Uji
error_test.sma <- test_ma.ts - data.gab[(n_train+1):n, 3]
error_test.dma <- test_ma.ts - data.gab2[(n_train+1):n, 6]
selisihsesopt  <- ramalanopt$mean - testing_ma$`T..degC`
selisihdesopt  <- ramalandesopt$mean - testing_ma$`T..degC`

tabel_akurasi <- data.frame(
  Metode = c("SMA (n=4)", "DMA (n=4)", "SES Optimum", "DES Optimum"),
  MSE  = c(mean(error_test.sma^2), mean(error_test.dma^2), mean(selisihsesopt^2), mean(selisihdesopt^2)),
  MAPE = c(mean(abs(error_test.sma/test_ma.ts)*100), mean(abs(error_test.dma/test_ma.ts)*100), 
          mean(abs(selisihsesopt/testing_ma$`T..degC`)*100), mean(abs(selisihdesopt/testing_ma$`T..degC`)*100))
)

  • SMA (n=4) menghasilkan MSE sebesar 0,1887 dan MAPE 33,84%. Artinya, metode ini memiliki tingkat kesalahan yang cukup rendah, tetapi masih lebih besar dibandingkan SES.
  • DMA (n=4) memiliki MSE 4,3940 dan MAPE 127,80%, yang merupakan nilai kesalahan paling besar. Jadi, DMA merupakan metode dengan kinerja paling buruk pada data ini.
  • SES Optimum memiliki MSE 0,0802 dan MAPE 23,04%, keduanya merupakan nilai terkecil dibandingkan metode lainnya. Jadi, SES Optimum merupakan metode terbaik dalam melakukan peramalan pada data tersebut.
  • DES Optimum menghasilkan MSE 0,7063 dan MAPE 47,93%. Kinerjanya lebih buruk dibandingkan SMA dan SES, tetapi masih lebih baik daripada DMA.

Berdasarkan nilai MSE dan MAPE, metode SES Optimum memberikan hasil peramalan terbaik karena memiliki nilai kesalahan paling rendah, yaitu MSE sebesar 0,0802 dan MAPE sebesar 23,04%. Sementara itu, DMA (n=4) memberikan hasil peramalan terburuk dengan MSE sebesar 4,3940 dan MAPE sebesar 127,80%. Dengan demikian, metode SES Optimum dipilih sebagai metode peramalan yang paling sesuai untuk data yang digunakan.

Kesimpulan:

  1. Metode Peramalan Terbaik: Metode Single Exponential Smoothing (SES) Optimum adalah metode terbaik untuk meramalkan suhu udara per jam, karena menghasilkan tingkat kesalahan (error) terkecil pada data uji:
  • MSE: 0,08018
  • MAPE: 23,04%
  1. Perbandingan Akurasi Seluruh Metode:
  • SES Optimum: MSE = 0,08018 | MAPE = 23,04%
  • SMA (\(n=4\)): MSE = 0,18870 | MAPE = 33,84%
  • DES Optimum: MSE = 0,70632 | MAPE = 47,93%
  • DMA (\(n=4\)): MSE = 4,39398 | MAPE = 127,80%
  1. Alasan Statistik: Data suhu per jam bersifat stasioner (berfluktuasi melandai di sekitar rata-rata -1.5°C) tanpa tren jangka panjang. Metode ber-tren ganda (DMA dan DES) memproyeksikan tren naik secara berlebihan (overshooting), sehingga menghasilkan kesalahan yang besar. Sebaliknya, metode tunggal (SES) mampu memotong pertengahan fluktuasi data aktual dengan sangat presisi.
